function getRelativeLuminance(hexColor) {
  const rgb = [r, g, b].map(ch => {
    const v = ch / 255
    return v <= 0.03928
      ? v / 12.92
      : ((v + 0.055) / 1.055) ** 2.4
  })
  return 0.2126*rgb[0]
       + 0.7152*rgb[1]
       + 0.0722*rgb[2]
}

function buildBackgroundGenerationPlan(input) {
  const words = tokenize(input.prompt)

  // Detecção de forma geométrica
  inferShapeFromPrompt(words)
  // "wave" → waves | "diamond" → diamonds
  // "flower"/"floral" → organic_floral

  // Detecção de tipo de composição
  inferCompositionFromPrompt(words, shape)
  // "flower" → radial_floral_cluster
  // "wave"   → wave_motion
  // "beam"   → beam_directional
  // "dense"  → structured_grid

  // Inferência de densidade
  inferDensityFromPrompt(words)
  // "minimal"/"clean" → minimal
  // "dense"/"rich"    → rich  (layering_depth=9)

  // Paleta a partir de cor detectada
  inferPaletteFromPrompt(words, palette)
  // "azul" → ['#1e3a8a','#3b82f6','#bfdbfe']
  // "roxo" → ['#3b0764','#9333ea','#e9d5ff']

  // Aplicação de hints de estilo/mood
  STYLE_HINTS = {
    editorial: { texture_mode: 'editorial_soft' },
    minimal:   { density: 'minimal', blur: 0.35 },
    geometric: { composition: 'structured_grid' },
    runway:    { composition: 'beam_directional' },
  }
  MOOD_HINTS = {
    dreamy:    { blur_strength: 0.85, glow: 0.76 },
    energetic: { density:'rich', glow: 0.92 },
    bold:      { contrast:'high', glow: 0.80 },
  }

  return {
    shape_language, composition_type,
    palette: [dark, mid, light],  // 3 cores
    density, layering_depth,      // 4–12 camadas
    glow_intensity, blur_strength,
    motion_direction, texture_mode,
  }
}

function generateProceduralBackground(plan, seed, prompt) {
  const random = createSeededRng(seed)  // determinístico
  const layerCount = clamp(plan.layering_depth, 3, 12)

  // Tight patterns: grid denso pré-definido
  detectTightPattern(plan)
  // arrows/waves/flowers/stars/circles
  // → buildTightPatternLayer() em vez de scatter

  // Para cada camada:
  for (let i = 0; i < layerCount; i++) {
    const shapes = buildShapeLayer(plan, random, i, layerCount)
    // shapes: circles, triangles, stars,
    //         diamonds, waves, beams...
    // posição: clusters + drift + motion alinhado
    //   orbital_field → orbital rings
    //   beam_directional → streaks verticais
    //   structured_grid → snap ao grid
  }

  // Composição SVG em camadas:
  // 1. linearGradient base (palette[0–2])
  // 2. N layers de shapes com blur (feGaussianBlur)
  // 3. Grain texture (noise pattern)
  // 4. Vinheta lateral esquerda
  // 5. Texto do prompt (opacidade 0.35)

  return `data:image/svg+xml;utf8,${encodeURIComponent(svg)}`
}

function renderFabricTextureToCanvas({ width, height, color, material }) {
  // 1. Preenche base com noise grain (varredura 2×2px)
  for (let y=0; y<height; y+=2)
    for (let x=0; x<width; x+=2)
      grain > 0.5 ? tint(color, +18, noise) : tint(color, -14, noise)

  // 2. Dispatch pelo tipo de material
  switch (material.type) {
    case 'embroidered_fabric': drawThreadField(ctx, ...)
    // threads em ângulo (0°, 90°, 45°, cross)
    // spacing = 34 - density*0.22   (3–31px)
    // jitter pseudoaleatório (x*13+y*7)%9
    // highlight: tint(+42 * contrastScale, alpha=0.30+emboss*0.32)
    // shadow:    tint(-36 * contrastScale, alpha=0.44+emboss*0.28)
    // cross: 2 passagens em ângulo + ângulo+π/2.5

    case 'lego_material': drawLegoField(ctx, ...)
    // blockSize = 56 - density*0.22  (22–54px)
    // arredondado (roundRect, radius=blockSize*0.15)
    // highlight superior, borda escura
    // 4 studs por bloco: radial gradient (#lighter→#darker)

    case 'water_material': drawWaterField(ctx, ...)
    // rippleGap = 36 - density*0.16  (10–34px)
    // onda seno: sin((x+y*0.6)/wave) * amplitude
    // segunda onda (offset 35%) + caustics elípticos

    case 'glass_material': drawGlassField(ctx, ...)
    // cellSize = 68 - density*0.22  (22–72px)
    // paneGradient linear (tint+26 → white → tint-12)
    // borda clara + linha diagonal de reflexo
  }

  // 3. Finish pass sobre o canvas completo
  if (material.finish === 'satin')
    // highlight diagonal: tint(+26, opacity 0.32) → white 0.04 → tint(-28, 0.48)
  else  // matte
    // gradiente suave: tint(+14, 0.18) → tint(-18, 0.28)

  // 4. Canvas decorativo: stitch border (opcional)
  if (material.stitchBorder)
    // setLineDash([9,7]) → roundRect com padding 5%
    // linha interna pontilhada [2,12]

  return { textureDataUrl, decorativeDataUrl }  // PNG data:URLs
}

// Determina qual área do card recebe a textura
function buildFabricScopeStyle(scope) {
  if (scope === 'hero_block')    return { inset: '8% 6% 45% 6%' }   // bloco superior
  if (scope === 'content_block') return { inset: '50% 6% 8% 6%' }   // bloco inferior
  return { inset: '0' }  // 'card' → todo o card (padrão)
}
